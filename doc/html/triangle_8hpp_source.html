43 bool hit(const Ray& r, Interval ray_t, CollisionRecord& rec) const override {
44 // If any vertex is at infinity, there is not intersection
45 if (is_at_infinity(skeleton[0]) || is_at_infinity(skeleton[1]) || is_at_infinity(skeleton[2])) return false;
46 point p0 = homogenize(moved_point(0, r.time()));
47 point p1 = homogenize(moved_point(1, r.time()));
48 point p2 = homogenize(moved_point(2, r.time()));
49
50 const double epsilon = 1e-8; // Handle near parralel intersections to make up for doubles nonsense
51
52 // Find the edges of the triangle as vectors
53 point e1 = p1 - p0;
54 point e2 = p2 - p0;
55
56 // Vector representation of intersection candidate
57 point pvec = cross(r.direction(), e2);
58
59 // Determine whether the ray is sufficiently near parralel
60 double det = dot(e1, pvec);
61 if (std::fabs(det) < epsilon) return false;
62 double inv_det = 1.0 / det; // for fractions
63
64 // Ray is not parrelel. Does it intersect?
65
66 // A point inside the triangle can be represented as P = A + u*e1 + v*e2 : u >= 0, v >= 0, u + v <= 1
67 point tvec = r.origin() - p0;
68 double u = dot(tvec, pvec) * inv_det;
69 point qvec = cross(tvec, e1);
70 double v = dot(r.direction(), qvec) * inv_det;
71 bool intersection = u >= 0 && v >= 0 && u + v <= 1;
72 if (!intersection) return false;
73
74 // Now we need to determine if the intersection is within the interval of the ray
75
76 double t = dot(e2, qvec) * inv_det;
77 bool intersects_in_range = ray_t.contains(t);
78 if (!intersects_in_range) return false;
79
80
81 // There is an intersection! Populate rec.
82 point normal = cross(e1, e2);
83 normal /= norm(normal);
84
85 rec.record(r, t, normal, this);
86 return true;
87
88 }

93 point2D get_tcoords(const point& p, double time) const override {

95 const point v0 = moved_point(0, time);
96 const point v1 = moved_point(1, time);
97 const point v2 = moved_point(2, time);
98
99 // Barycentric Coordinates
100 const double area = (v1[X] - v0[X]) * (v2[Y] - v0[Y]) - (v1[Y] - v0[Y]) * (v2[X] - v0[X]);
101 if (std::fabs(area) < 1e-12) return point2D(0, 0);
102 const double b0 = ((v1[X] - p[X]) * (v2[Y] - p[Y]) - (v1[Y] - p[Y]) * (v2[X] - p[X])) / area;
103 const double b1 = ((v2[X] - p[X]) * (v0[Y] - p[Y]) - (v2[Y] - p[Y]) * (v0[X] - p[X])) / area;
104
105 // Interpolate the texture based on barycentric coordinates
106 return b0 * t_coords[0] + b1 * t_coords[1] + (1.0 - b0 - b1) * t_coords[2];

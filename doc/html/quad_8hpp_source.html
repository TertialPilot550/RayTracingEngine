20 // Find the object at this particular moment in time
21 const point origin = moved_point(0, r.time());
22 const point edge_u = as_vector(motion.transform(as_point(u), r.time()));
23 const point edge_v = as_vector(motion.transform(as_point(v), r.time()));
24 const point n = cross(edge_u, edge_v);
25
26 // Make sure that the normal vector is valid
27 const double n_length = norm(n);
28 if (n_length < 1e-12) return false;
29 const point normal = n / n_length;
30
31 // Barycentric Coordinates
32 const double denom = dot(normal, r.direction());
33 if (std::fabs(denom) < 1e-8) return false;
34 const double t = (dot(normal, origin) - dot(normal, r.origin())) / denom;
35 if (!ray_t.contains(t)) return false;
36 const point q = r.at(t) - origin;
37 const double uu = dot(edge_u, edge_u);
38 const double vv = dot(edge_v, edge_v);
39 const double a = dot(q, edge_u) / uu;
40 const double b = dot(q, edge_v) / vv;
41
42 // If the point is not inside the quadrilateral, return false
43 if (a < 0.0 || a > 1.0 || b < 0.0 || b > 1.0) return false;
44
45 // Hit! Populate the record and return
